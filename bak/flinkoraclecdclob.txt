1.2. MySQL 触发器
通过触发器解决 blob 字段更新后被置为 __debezium_unavailable_value 值问题：
CREATE TRIGGER trg_block_debezium_placeholder_blob
BEFORE UPDATE ON flink_cdc.CDC_SINK
FOR EACH ROW
BEGIN
    -- 将传进来的 BLOB 转换为字符串，判断是否包含占位符
    IF CONVERT(NEW.IMG USING utf8mb4) LIKE '%__debezium_unavailable_value%' THEN
        -- 强行把新值还原为原本存在数据库里的老图片，忽略这次对 IMG 的更新
        SET NEW.IMG = OLD.IMG;
    END IF;
END

2.2. DM 触发器
-- 1. 创建达梦目标表
CREATE TABLE FLINKUSER.CDC_SINK (
  ID INT PRIMARY KEY,
  NAME VARCHAR(255),
  IMG BLOB
);

-- 2. 创建达梦高级 PL/SQL 触发器
CREATE OR REPLACE TRIGGER FLINK_CDC_TEST.TRG_BLOCK_DEBEZIUM_PLACEHOLDER_BLOB
BEFORE UPDATE ON FLINK_CDC_TEST.CDC_SINK
FOR EACH ROW
DECLARE
    V_PLACEHOLDER RAW(100);
BEGIN
    -- 将占位符文本转换为达梦可以对 BLOB 进行检索的 RAW 二进制格式
    V_PLACEHOLDER := UTL_RAW.CAST_TO_RAW('__debezium_unavailable_value');

    -- 使用 DBMS_LOB.INSTR 函数查找新传入的 :NEW.IMG 中是否包含该占位符
    -- 如果返回值 > 0，说明上游发送的是未变更的大字段占位符
    IF DBMS_LOB.INSTR(:NEW.IMG, V_PLACEHOLDER, 1, 1) > 0 THEN
        -- 【核心魔法】：强行把新值替换回达梦里原有的老图片，拒绝被占位符覆盖
        :NEW.IMG := :OLD.IMG;
    END IF;
END;
